\begin{frame}{1. Introduction \& Background Study}
    \begin{block}{What is Data Compression?}
        \begin{itemize}
            \item Process of reducing physical storage footprint and transmission bandwidth of digital files.
            \item \textbf{Lossless Compression}: Guarantees 100\% exact bit-for-bit reconstruction of original data without distortion or data loss.
        \end{itemize}
    \end{block}
    
    \vspace{0.3cm}
    \begin{block}{Theoretical Foundation}
        \begin{itemize}
            \item \textbf{Shannon Entropy (1948)}: Theoretical minimum limit for average bits per symbol:
            $$H(X) = - \sum_{i=1}^n P(x_i) \log_2 P(x_i)$$
            \item \textbf{Huffman Algorithm (1952)}: Greedy prefix-coding algorithm that assigns shorter codes to frequent symbols, achieving optimal prefix compression.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{4. Methodology \& Working Mechanism}
    \begin{block}{Five Steps of Huffman Compression}
        \begin{enumerate}
            \item \textbf{Frequency Profiling}: Count occurrences of all 256 byte values ($0\text{x}00$ to $0\text{xFF}$).
            \item \textbf{Min-Heap Initialization}: Insert character nodes ordered by frequency.
            \item \textbf{Tree Construction}: Repeatedly merge lowest two nodes until one root remains.
            \item \textbf{Code Assignment}: Left branch = `0', Right branch = `1'; extract variable-length bitstrings.
            \item \textbf{Bitstream Packing}: Pack bit sequences into 8-bit bytes with header frequency metadata.
        \end{enumerate}
    \end{block}

    \vspace{0.2cm}
    \begin{block}{Decompression Workflow}
        \begin{itemize}
            \item Read frequency table $\rightarrow$ Reconstruct exact binary tree $\rightarrow$ Traverse bits from root to leaf $\rightarrow$ Output original bytes $\rightarrow$ Verify SHA-256 checksum.
        \end{itemize}
    \end{block}
\end{frame}
